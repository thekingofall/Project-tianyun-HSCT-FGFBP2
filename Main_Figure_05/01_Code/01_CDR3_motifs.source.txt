import numpy as np
import pandas as pd
import weblogo
from weblogo import ColorScheme, SymbolColor

import IPython.display

seqs = ["CASSYGQKGQPQHF", "CASSQASRGDTQYF", "CASSERGPEQFYGYTF", "CASSPTMNTEAFF", "CASGPDGAYEQYF", "CASSKTSGPYNEQFF", "CASSLESLGRTSTDTQYF", "CASSLLTQEGEQFF", "CASSTLAGGTDTQYF", "CASSLLPGDDGTNEKLFF", "CASSVLAGGTDTQYF", "CASSPGQGTRNTEAFF", "CASSLGAEAFF", "CASSQSTAANEKLFF", "CASSMFVQGSGANVLTF", "CAITSGRDRTDTQYF", "CASSQESGAGYGYTF", "CASSLRGNEQYF", "CASSLSGGGGPNTEAFF", "CASSQDGRARNEQFF"]


import palmotif

motif, s = palmotif.compute_pal_motif(seqs[0], seqs)

alphabet = weblogo.seq.Alphabet(''.join(motif.index))
hydrophobicity = ColorScheme([SymbolColor("RKDENQ", "#0985D4", "hydrophilic"),
                              SymbolColor("SGHTAP", "#FDB400", "neutral"),
                              SymbolColor("YVMCLFIW", "#9F4693", "hydrophobic"),
                              SymbolColor("-*ZX", "gray", "default")],
                             alphabet=alphabet)

logodata = weblogo.LogoData(length=motif.shape[1],
                            alphabet=alphabet,
                            entropy=motif.sum(axis=0),
                            weight=np.ones(motif.shape[1]),
                            counts=motif.values.T)

logooptions = weblogo.LogoOptions(  resolution=200,
                                    show_fineprint=False,
                                    yaxis_scale=1.1*np.max(motif.values),
                                    color_scheme=hydrophobicity,
                                    unit_name='nats',
                                    yaxis_label='bits')
logoformat = weblogo.LogoFormat(logodata, logooptions)
logobytes = weblogo.logo_formatter.png_formatter(logodata, logoformat)
IPython.display.Image(data=logobytes)

output_filename = "CD8teff3.png"  # 可自行修改文件名

with open(output_filename, "wb") as f:
  f.write(logobytes)

print(f"PNG 图片已成功保存为: {output_filename}")
